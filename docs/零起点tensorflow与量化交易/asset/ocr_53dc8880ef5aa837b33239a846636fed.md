```
def lstm010(num_in,num_out=1):
model = Sequential()
#
model.add(LSTM(num_in*8, input_shape=(num_in,1)))
model.add(layers.Dense(1))
#
model.compile(loss='mse',optimizer='rmsprop', metrics=['accuracy' ])
#
return model
```
