#定义MeanVariance 类
from scipy import linag
class MeanVariance:
#定义构造器，传入收益率数据
def __init__(self,returns):
self.retums=returns
#定义最小化方差的函数，即求解二次规划
def minVar(self,goalRet):
covs=np.array(self.retums.cov())
means=np.array(self.retums.mean())
L1=np.append(np.append(covs.swpaces(0,1,[means],0),
[np.ones(len(means))],0).swpaces(0,1)
L2=list(np.ones(len(means)))
L2.extend([0,0])
L3=list(means)
L3.extend([0,0])
L4=np.array([L2,L3])
L=np.append(L1,L4,0)
result=linag.solve(L,np.append(np.zeros(len(means)),[1,goalRet],0))
return(np.array([list(self.retums.columns),result[:-2]]))
#定义绘制最小方差前缘曲线函数
def frontierCurve(self):
goals=[x/500000 for x in range(-100,4000)]
variances=list(map(lamba x: self.calVar(self.minVar(x[1:]),astype(np.float)),
goals))
plt.plot(variances,goals)
