elif method=='Cointegration':
            spread=self.CointegrationSpread(priceX,priceY,formPeriod,formPeriod)
            mu=np.mean(spread)

            UpperBound=mu+width+sd
            LowerBound=mu-width+sd
            return(UpperBound,LowerBound)
