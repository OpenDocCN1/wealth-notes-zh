d2 = d1 - vol*sqrt(maturity)

if call_or_put==1:

    call_price = spot*norm.cdf(d1) - strike*exp(-r*maturity)*norm.cdf(d2)

    return call_price

elif call_or_put==0:

    put_price=strike*exp(-r*maturity)*norm.cdf(-d2) - spot*norm.cdf(-d1)

    return put_price
