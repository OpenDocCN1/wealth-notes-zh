def cancelOrder(self, vtOrderID):
    """撤单"""
    # 如果发单号为空字符串，则不进行后续操作
    if not vtOrderID:
        return

    if STOPORDERPREFIX in vtOrderID:
        self.ctaEngine.cancelStopOrder(vtOrderID)
    else:
        self.ctaEngine.cancelOrder(vtOrderID)
