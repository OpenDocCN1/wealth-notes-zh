self.xminBar.openInterest = bar.openInterest
self.xminBar.volume += int(bar.volume)

# x分钟已经走完
if not (bar.datetime.minute + 1) % self.xmin:  # 可以用x整除
    # 生成上一x分钟K线的时间戳
    self.xminBar.datetime = self.xminBar.datetime.replace(second=0,
